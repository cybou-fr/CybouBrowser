cmake_minimum_required(VERSION 3.24)
project(CybouBrowser VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Quick QuickControls2 WebEngineQuick WebEngineCore Network LinguistTools)
qt_standard_project_setup(REQUIRES 6.9)
qt_add_executable(cybou src/main.cpp src/browserstore.h src/security/NetworkPolicy.h src/security/SecurityTests.h)
if(WIN32)
    target_sources(cybou PRIVATE assets/cybou.rc)
endif()
qt_add_resources(cybou "brand" PREFIX "/" FILES assets/cybou.png assets/speaker.svg assets/speaker-muted.svg)
file(GLOB CONTROL_BAR_ICONS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "assets/controlbar/*.svg")
qt_add_resources(cybou "controlbar_icons" PREFIX "/" FILES ${CONTROL_BAR_ICONS})
qt_add_qml_module(cybou URI Cybou VERSION 1.0
    SOURCES src/controlbar/ControlBarController.h src/controlbar/ControlBarController.cpp src/controlbar/CapabilityModel.h src/controlbar/RequestObserver.h
    QML_FILES qml/Main.qml qml/NavButton.qml qml/SettingsPanel.qml qml/ControlBar.qml qml/ControlBarIcon.qml qml/CapabilityPopover.qml qml/ControlBarPositionPicker.qml qml/ControlBarHarness.qml)
target_include_directories(cybou PRIVATE src/controlbar)
qt_add_translations(cybou TS_FILES translations/cybou_ru.ts RESOURCE_PREFIX "/translations")
target_link_libraries(cybou PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick Qt6::WebEngineCore Qt6::Network)
set_target_properties(cybou PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
install(TARGETS cybou BUNDLE DESTINATION . RUNTIME DESTINATION bin)
