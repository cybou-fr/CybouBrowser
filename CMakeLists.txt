cmake_minimum_required(VERSION 3.24)
project(CybouBrowser VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Quick QuickControls2 WebEngineQuick Network)
qt_standard_project_setup(REQUIRES 6.9)
qt_add_executable(cybou src/main.cpp src/browserstore.h)
if(WIN32)
    target_sources(cybou PRIVATE assets/cybou.rc)
endif()
qt_add_resources(cybou "brand" PREFIX "/" FILES assets/cybou.png)
qt_add_qml_module(cybou URI Cybou VERSION 1.0 QML_FILES qml/Main.qml qml/NavButton.qml qml/SettingsPanel.qml)
target_link_libraries(cybou PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick Qt6::Network)
set_target_properties(cybou PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
install(TARGETS cybou BUNDLE DESTINATION . RUNTIME DESTINATION bin)
